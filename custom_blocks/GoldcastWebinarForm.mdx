export const GoldcastWebinarForm = () => {
  const EVENT_ID = "fefc2076-a74f-4126-9ad4-e8e84411d1d3";
  const SCRIPT_SRC = "https://form-embed-sdk.goldcast.io/embed.js";

  const setup = (node) => {
    if (!node || node.dataset.started) return;
    node.dataset.started = "yes";

    const status = node.querySelector("#gc-status");
    const say = (msg) => {
      if (status) status.textContent = msg;
    };

    // Add _Days=yes to the URL so the hidden field fills in
    const url = new URL(window.location.href);
    if (!url.searchParams.has("_Days")) {
      url.searchParams.set("_Days", "yes");
      window.history.replaceState({}, "", url);
    }

    const start = () => {
      try {
        new window.GOLDCAST.RegistrationWidget({
          event_id: EVENT_ID,
          mode: "embed",
          containerSelector: "#goldcast-registration-form",
          theme_obj: {},
          backendUrl: "https://backend.goldcast.io",
        });
        say("");
      } catch (e) {
        say("Widget error: " + e.message);
      }
    };

    if (window.GOLDCAST) {
      start();
      return;
    }

    say("Loading form...");
    const script = document.createElement("script");
    script.src = SCRIPT_SRC;
    script.async = true;
    script.onload = () => start();
    script.onerror = () => say("The form couldn't load. Refresh the page, or email support@recurly.com.");
    document.head.appendChild(script);
  };

  return (
    <div id="register" className="rc-form-outer">
      <style>{`
        .rc-form-outer {
          max-width: 1120px;
          margin: 0 auto;
          padding: 0 32px;
          box-sizing: border-box;
          font-family: "Figtree", "Helvetica Neue", Helvetica, arial, sans-serif;
        }
        .rc-form-outer .rc-form-eyebrow {
          display: inline-block;
          font-size: 12px;
          font-weight: 800;
          letter-spacing: 1.4px;
          text-transform: uppercase;
          color: #FF5126;
          margin: 12px 0 8px;
        }
        .rc-form-outer .rc-form-title {
          font-size: 2rem;
          font-weight: 800;
          line-height: 1.2;
          color: #0D0D0B;
          margin: 0 0 22px;
        }
        .rc-form-outer .rc-form-box {
          position: relative;
          background: #FFFFFF;
          border: 2px solid #0D0D0B;
          border-radius: 16px;
          padding: 30px 28px 28px;
          min-height: 420px;
          box-sizing: border-box;
          box-shadow: 0 14px 36px rgba(13, 13, 11, 0.14);
        }
        .rc-form-outer .rc-form-box::before {
          content: "";
          position: absolute;
          left: -2px;
          right: -2px;
          top: -2px;
          height: 8px;
          background: #FFD706;
          border-radius: 16px 16px 0 0;
        }
        .rc-form-outer #gc-status {
          color: #807D75;
          font-size: 14px;
          margin: 0 0 12px;
        }
        @media (max-width: 768px) {
          .rc-form-outer { padding: 0 20px; }
          .rc-form-outer .rc-form-title { font-size: 1.6rem; }
          .rc-form-outer .rc-form-box { padding: 24px 18px 20px; }
        }
      `}</style>

      <span className="rc-form-eyebrow">Register</span>
      <h2 className="rc-form-title">Save your seat</h2>
      <div ref={setup} className="rc-form-box">
        <p id="gc-status">Loading form...</p>
        <div id="goldcast-registration-form" />
      </div>
    </div>
  );
};
